　ゲームシーケンスはC言語のメイン関数でおよそ次のような形をしています。
Uint32 SMMA_MainMode;             /*　ゲームモード変数　　　　　　　*/
Uint32 SMMA_MainLevel;            /*　ゲームモード内のレベル　　　　*/
void main(void)
{
    SMMA_IniSystem();             /*　システム初期化　　　　　　　　*/
    SMV1_SprCmdStart();
    SMV1_SprCmdEnd();
    SCL_DisplayFrame();
    SMMA_MainMode = LOGO_MODE;
    SMMA_MainLevel = 0;
    for(;;) {                     /*　無限ループ　　　　　　　　　　*/
          switch(SMMA_MainMode) { /*　ゲームモードが？　　　　　　　*/
          case LOGO_MODE:         /*　セガロゴ状態のとき　　　　　　*/
              SMLO_SegaLogo();    /*　セガロゴ表示　　　　　　　　　*/
              break;
          case TITLE_MODE:        /*　タイトル状態のとき　　　　　　*/
              SMTI_Title();       /*　タイトル表示　　　　　　　　　*/
              break;
          case SELECT_MODE:       /*　プログラムセレクト状態のとき　*/
              SMSL_Select();      /*　プログラムセレクト実行　　　　*/
              break;
          }
    }
}

　メイン関数内ではゲームモード変数によってサブルーチン関数を呼び出すことを繰り返しているだけです。ゲームモード変数は次の動作状態を決定する変数です。ゲームモード変数は外部変数として定義されているため、どのサブルーチンからでも状態を書き換えることができます。初期値としてLOGO_MODEをもっているのでセガロゴ表示からスタートします。以降にそれぞのサブルーチンで行っている処理内容を示します。
