アクションコントロール使用例

　実際にアクションの使用例を示します。

/*
 *    変数定義
 */
Uint8  Mainlevel;
Uint8  Gameover;
SMTA_DefActWk(GAMEACT,
    SPRITE sprite;                     /* SPRITE DISPLAY　　　　　　          */
    Sint16 wreg[4];                    /* 汎用レジスタ                        */
    Sint16 wcnt[4];                    /* 汎用カウンタ                        */
    Uit8 colino;                       /* collision table index   　　　　　　*/
    Uint8 coliatr;                     /* collision attribute    　 　　　　　*/
    Uint8 coliflg;                     /* collision flag         　 　　　　　*/
    Uint8 atp;                         /* kougeki ryoku           　　　　　　*/
    Sint16 hp;                         /* hit point               　　　　　　*/
);                                     /* 専用アクション構造体GAMEACTを宣言　  */
/*
 *    関数プロトタイプ
 */
void GameMain(void);
void PlayerAct(GAMEACT *);
void EnemyAct(GAMEACT *);
void ScrollAct(GAMEACT *);
/* ####[ アクション使用例  メイン部分 ]#### */
void GameMain(void)
{
    enum {
        INIT, MAIN,
    };
    Mainlevel = INIT;
    Gameover = 0;                      /* ゲーム終了のフラグ　　　　　　　　　　*/
    for(;;) {
        intWait();                     /* Ｖブランク待ち　　　　　　　　　　　　*/
        switch(Mainlevel) {
        case INIT:
            InitVdp();                 /* ＶＤＰの初期化　　　　　　　　　　　　*/
            SMTA_ActWkInit();          /* アクションワークの初期化　　　　　　　*/
            SMTA_MakeAction(PlayerAct);
                                       /* プレイヤーのアクションセット　　　　　*/
            SMTA_MakeAction(EnemyAct);
                                       /* エネミーのアクションセット　　　　　　*/
            SMTA_MakeAction(ScrollAct);
                                       /* 背景のアクションセット　　　　　　　　*/
            Mainlevel++;
        case MAIN:
            SMTA_ActionLoop();
            if (Gameover) Mainlevel++;
            break;
        case EXIT:
            return;                    /* ゲーム終了　　　　　　　　　　　　　　*/
        }
    }
