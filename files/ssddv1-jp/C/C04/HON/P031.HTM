        　ポインティングデバイスを示すデータ型です。

        typedef struct  {  /* ポインティングデバイスデータ型      */
  PerPntData dgt;  /* デジタルデバイスデータ型　　　　    */
  Uint16 data;     /* データ　　　　　　　　　　　　　    */
  Sint16 x;        /* Ｘ軸移動量（－128～127）　　　　　　*/
  Sint16 y;        /* Ｙ軸移動量（－128～127）　　　　　　*/
}PerPntData;
