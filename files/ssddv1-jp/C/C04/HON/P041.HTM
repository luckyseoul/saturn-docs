        typedef  struct  BupStat {
  Uint32  totalsize;  /*　全容量()　　　　　　　*/
  Uint32  totalblock; /*　ブロック数　　　　　　*/
  Uint32  blocksize;  /*　１ブロックのサイズ()　*/
