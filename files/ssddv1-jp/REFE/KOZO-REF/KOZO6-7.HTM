回転パラメータ

構造

typedef   struct    {
　　　　　          FIXED XST ;
　　　　　          FIXED YST ;
　　　　　          FIXED ZST ;
　　　　　          FIXED DXST ;
　　　　          　FIXED DYST ;
　　　　　          FIXED DX ;
　　　　          　FIXED DY ;
　　　　          　FIXED MATA ;
　　　　          　FIXED MATB ;
　　　　　          FIXED MATC ;
　　　　　          FIXED MATD ;
　　　　          　FIXED MATE ;
　　　　          　FIXED MATF ;
　　　　          　Sint16 PX ;
　          　　　　Sint16 PY ;
　　　　　          Sint16 PZ ;
　　　　　          Sint16 dummy0 ;
　　　　　          Sint16 CX ;
　　　　          　Sint16 CY ;
　　　          　　Sint16 CZ ;
　　　          　　Sint16 dummy1 ;
　　　　　          FIXED MX ;
　　　　　          FIXED MY ;
　　　　          　FIXED KX ;
　　　          　　FIXED KY ;
　　　          　　Uint32 KAST ;
　　　          　　Sint32 DKAST ;
　　          　　　Sint32 DKA ;
          } ROTSCROLL;

メンバー

XST       スクロール画面スタート座標　Xst
YST       スクロール画面スタート座標　Yst
ZST       スクロール画面スタート座標　Zst
DXST      スクロール画面垂直方向座標増分　dXst
DYST      スクロール画面垂直方向座標増分　dYst
DX        スクロール画面水平方向座標増分　dX
DY        スクロール画面水平方向座標増分　dY
MATA      回転マトリクスパラメータＡ
MATB      回転マトリクスパラメータＢ
MATC      回転マトリクスパラメータＣ
MATD      回転マトリクスパラメータＤ
MATE      回転マトリクスパラメータＥ
MATF      回転マトリクスパラメータＦ
PX        視点座標　Px
PY        視点座標　Py
PZ        視点座標　Pz
dummy0    ダミー領域
CX        中心座標　Px
CY        中心座標　Py
CZ        中心座標　Pz
dummy1    ダミー領域
MX        平行移動量　Mx
MY        平行移動量　My
KX        拡大縮小係数　Kx
KY        拡大縮小係数　Ky
KAST      係数テーブルスタートアドレス　KAat
DKAST     係数テーブル垂直方向アドレス増分　DKAst
DKA       係数テーブル水平方向アドレス増分　DKA

解説

回転パラメータテーブルを定義します。
回転スクロール画面は、回転パラメータテーブルをラインごとに読み込み、その値にしたがって画面を表示します。

備考

定義されている２つのダミー領域は、アドレス調整のためのもので、使用することはできません。
