; Required: Required
; Selection: Selectable or not required

Define dirsmpdisc. \ Sample \; selectable
Disc sample1.DSK; required
Session SEMIXA; Necessary
LeadIn MODE1; required
EndLeadIn; required −
;
SystemArea [dirsmpdisc] sys_ip.bin; Required
;
Track MODE1; required
    Volume ISO9660 sample1.PVD; required
    PrimaryVolume 00:02:16; required
    SystemIdentifier "SEGA SEGASATURN"; required
    VolumeIdentifier "SAMPLE_GAME_TITLE"; Required
    VolumeSetIdentifier "SAMPLE_GAME_TITLE"; Required
    PublisherIdentifier "SEGA ENTERPRISES, LTD."; Required
    DataPreparerIdentifier "SEGA ENTERPRISES, LTD."; Required
    CopyrightFileIdentifier "SMP_CPY.TXT"; required
    AbstractFileIdentifier "SMP_ABS.TXT"; Required
    BibliographicFileIdentifier "SMP_BIB.TXT"; Required
    VolumeCreationDate 22/11/1994 00: 01: 02: 00: 36; selectable
    VolumeModificationDate 22/11/1994 00: 01: 02: 00: 36; selectable
    EndPrimaryVolume; required −
    EndVolume; required −
;
    File SMP_CPY.TXT; required
    FileSource [dirsmpdisc] smp_cpy.txt; Required
    EndFileSource; required −
    EndFile; required −
    File SMP_ABS.TXT; Required
    FileSource [dirsmpdisc] smp_abs.txt; Required
    EndFileSource; required −
    EndFile; required −
    File SMP_BIB.TXT; required
    FileSource [dirsmpdisc] smp_bib.txt; Required
    EndFileSource; required −
    EndFile; required −
;
    File FILE0.BIN; selectable
    FileSource [dirsmpdisc] file0.bin; selectable
    EndFileSource; Select
    EndFile; Select
    ;
    ; File ~ EndFile; selectable
    ;
    PostGap 75; Necessary
EndTrack; required
;
