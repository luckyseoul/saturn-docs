付録A・各種サンプルリスト-3

;=======================================================================
;        sample1.scr -- CD-ROM XA        (Ver.1994-11-11)
;Note:  CD-ROM XA(MODE1 + MODE2 + CD-DA)ディスクのサンプルスクリプトです。
;       VCDPRE,VCDBUILD のバージョンは、Ver.3.10以降をご使用ください。
;    必: 必須
;    選: 選択可、無くても良い
;    不: パラメータの変更不可(このまま変更せずに使用してください。)
;    可: パラメータの変更可
;    −: パラメータなし
;    各行の先頭の１単語は、コマンド名です。このまま使用してください。
;=======================================================================
Define        dirsmpdisc.¥sample¥                              ;  選  可
Disc          sample1.DSK                                      ;必    可
Session       SEMIXA                                           ;必  不
LeadIn        MODE1                                            ;必  不
EndLeadIn                                                      ;必  −
;
SystemArea    [dirsmpdisc]sys_ip.bin                           ;必    可
;
Track          MODE1                                           ;必  不
    Volume                      ISO9660 sample1.PVD            ;必    可
    PrimaryVolume                  00:02:16                    ;必  不
    SystemIdentifier               "SEGA SEGASATURN"           ;必  不
    VolumeIdentifier               "SAMPLE_GAME_TITLE"         ;必    可
    VolumeSetIdentifier            "SAMPLE_GAME_TITLE"         ;必    可
    PublisherIdentifier            "SEGA ENTERPRISES,LTD."     ;必    可
    DataPreparerIdentifier         "SEGA ENTERPRISES,LTD."     ;必    可
    CopyrightFileIdentifier        "SMP_CPY.TXT"               ;必    可
    AbstractFileIdentifier         "SMP_ABS.TXT"               ;必    可
    BibliographicFileIdentifier    "SMP_BIB.TXT"               ;必    可
    VolumeCreationDate        22/11/1994 00:01:02:00:36        ;  選  可
    VolumeModificationDate    22/11/1994 00:01:02:00:36        ;  選  可
    EndPrimaryVolume                                           ;必  −
    EndVolume                                                  ;必  −
;
    File            SMP_CPY.TXT                                ;必    可
    FileSource        [dirsmpdisc]smp_cpy.txt                  ;必    可
    EndFileSource                                              ;必  −
    EndFile                                                    ;必  −
    File            SMP_ABS.TXT                                ;必    可
    FileSource        [dirsmpdisc]smp_abs.txt                  ;必    可
    EndFileSource                                              ;必  −
    EndFile                                                    ;必  −
    File            SMP_BIB.TXT                                ;必    可
    FileSource        [dirsmpdisc]smp_bib.txt                  ;必    可
    EndFileSource                                              ;必  −
    EndFile                                                    ;必  −
;
;  !!!補足!!!
;  1stREAD FILE の定義：ファイル名がアスキーコードで昇順
;
        File            0MAIN.BIN                              ;  選  可
            FileSource        [dirsmpdisc]file0.bin            ;  選  可
            EndFileSource                                      ;  選  −
        EndFile                                                ;  選  −
    ;
    ;        File 〜  EndFile                                  ;  選  可
    ;
    PostGap          75                                        ;必  不
EndTrack                                                       ;必  −
;
  Track        MODE2                                           ;必  不
    PreGap          150                                        ;必  不
    Extent                                                     ;必  −
  FileInterleave        1 3                                    ;  選  可
    File            INTFILE0.BIN                               ;  選  可
    FileSource        [dirsmpdisc]intfile0.bin                 ;  選  可
    EndFileSource                                              ;  選−
    EndFile                                                    ;  選−
    EndFileInterleave                                          ;  選−
    FileInterleave        1 3                                  ;  選  可
    File            INTFILE1.BIN                               ;  選  可
    FileSource        [dirsmpdisc]intfile1.bin                 ;  選  可
    EndFileSource                                              ;  選−
    EndFile                                                    ;  選−
  EndFileInterleave                                            ;  選−
  FileInterleave        1 3                                    ;  選  可
    File            INTFILE2.BIN                               ;  選  可
    FileSource        [dirsmpdisc]intfile2.bin                 ;  選  可
    EndFileSource                                              ;  選−
    EndFile                                                    ;  選−
  EndFileInterleave                                            ;  選−
  FileInterleave        1 3                                    ;  選  可
    File            INTFILE3.BIN                               ;  選  可
    FileSource        [dirsmpdisc]intfile3.bin                 ;  選  可
    EndFileSource                                              ;  選−
    EndFile                                                    ;  選−
  EndFileInterleave                                            ;  選−
  EndExtent                                                    ;必  −
    PostGap  150                                               ;必  不
EndTrack                                                       ;必  −
;
Track          CDDA                                            ;必  不
    Pause              150                                     ;必  不
    FileSource          [dirsmpdisc]sound0.da                  ;必    可
    EndFileSource                                              ;必  −
EndTrack                                                       ;必  −
;
;   !!!重要!!!
;   Track 〜  EndTrack                                         ;  選  可
;      CDDAトラックを複数トラック使用する場合は、全 CDDAトラック
;      の先頭に 「Pause   150」が必要です。
;
;
;        Track 〜 EndTrack                                     ;  選  可
;
LeadOut          CDDA                                          ;必  不
Empty            500                                           ;必  不
EndLeadOut                                                     ;必  −
EndSession                                                     ;必  −
EndDisc                                                        ;必  −
;====== End of file ====================================================
